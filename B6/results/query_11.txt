-- ---------------------------------------------------------------------
-- [Q11] 집계 (COUNT, SUM, AVG) : 결제 수단별 결제 건수, 합계, 평균 금액을 확인한다.
-- ---------------------------------------------------------------------
SELECT method,
       COUNT(*)           AS payment_count,
       SUM(amount)        AS total_amount,
       ROUND(AVG(amount)) AS avg_amount
FROM payment
GROUP BY method
ORDER BY total_amount DESC;

-- ---------------------------------------------------------------------
-- 실행 결과 (MySQL 8.0)
-- ---------------------------------------------------------------------
+--------+---------------+--------------+------------+
| method | payment_count | total_amount | avg_amount |
+--------+---------------+--------------+------------+
| CARD   |            14 |        99800 |       7129 |
| PAY    |             3 |        21800 |       7267 |
| CASH   |             5 |        16600 |       3320 |
+--------+---------------+--------------+------------+
