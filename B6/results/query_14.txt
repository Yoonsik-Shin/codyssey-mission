-- ---------------------------------------------------------------------
-- [Q14] DELETE : 취소된(CANCELLED) 구매 기록을 삭제한다.
--       orders를 먼저 지우면 FK 때문에 거부되므로 자식 테이블부터 지운다.
--       순서 : payment → order_item → orders
--       세 테이블이 함께 지워져야 하므로 하나의 트랜잭션으로 묶는다.
-- ---------------------------------------------------------------------
-- 삭제 전
SELECT status, COUNT(*) AS order_count
FROM orders
GROUP BY status
ORDER BY status;

START TRANSACTION;

DELETE FROM payment
WHERE order_id IN (SELECT order_id FROM orders WHERE status = 'CANCELLED');

DELETE FROM order_item
WHERE order_id IN (SELECT order_id FROM orders WHERE status = 'CANCELLED');

DELETE FROM orders
WHERE status = 'CANCELLED';

COMMIT;

-- 삭제 후
SELECT status, COUNT(*) AS order_count
FROM orders
GROUP BY status
ORDER BY status;

-- ---------------------------------------------------------------------
-- 실행 결과 (MySQL 8.0)
-- ---------------------------------------------------------------------
+-----------+-------------+
| status    | order_count |
+-----------+-------------+
| CANCELLED |           2 |
| PAID      |          17 |
| REFUNDED  |           1 |
+-----------+-------------+
+----------+-------------+
| status   | order_count |
+----------+-------------+
| PAID     |          17 |
| REFUNDED |           1 |
+----------+-------------+
