-- ---------------------------------------------------------------------
-- [Q13] UPDATE : 강남점에 제육볶음 도시락 20개가 입고되어 재고를 0 → 20으로 늘린다.
--       updated_at은 ON UPDATE CURRENT_TIMESTAMP(MySQL 전용) 덕분에 자동으로 갱신된다.
-- ---------------------------------------------------------------------
-- 수정 전
SELECT inventory_id, store_id, product_id, quantity, updated_at
FROM inventory
WHERE store_id = 1 AND product_id = 6;

UPDATE inventory
SET quantity = quantity + 20
WHERE store_id = 1
  AND product_id = 6;

-- 수정 후
SELECT inventory_id, store_id, product_id, quantity, updated_at
FROM inventory
WHERE store_id = 1 AND product_id = 6;

-- ---------------------------------------------------------------------
-- 실행 결과 (MySQL 8.0)
-- ---------------------------------------------------------------------
+--------------+----------+------------+----------+---------------------+
| inventory_id | store_id | product_id | quantity | updated_at          |
+--------------+----------+------------+----------+---------------------+
|            5 |        1 |          6 |        0 | 2026-11-02 12:30:00 |
+--------------+----------+------------+----------+---------------------+
+--------------+----------+------------+----------+---------------------+
| inventory_id | store_id | product_id | quantity | updated_at          |
+--------------+----------+------------+----------+---------------------+
|            5 |        1 |          6 |       20 | 2026-11-16 04:53:51 |
