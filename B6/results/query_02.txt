-- ---------------------------------------------------------------------
-- [Q02] 기본 조회 : 가장 비싼 상품 5개를 확인한다.
--       LIMIT은 MySQL 전용 문법이다. (표준 SQL은 FETCH FIRST 5 ROWS ONLY)
-- ---------------------------------------------------------------------
SELECT product_id, name, price
FROM product
ORDER BY price DESC
LIMIT 5;

-- 실행 결과 (MySQL 8.0)

|         11 | 건전지 AA 4입            |  4800 |
|          4 | 봉지라면 5입             |  4500 |
