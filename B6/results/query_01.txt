-- ---------------------------------------------------------------------
-- [Q01] 기본 조회 : 가격이 2,000원 이상 5,000원 이하인 상품을 싼 순서로 확인한다.
-- ---------------------------------------------------------------------
SELECT product_id, name, price
FROM product
WHERE price BETWEEN 2000 AND 5000
ORDER BY price, name;
