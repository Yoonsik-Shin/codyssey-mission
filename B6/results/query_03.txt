-- ---------------------------------------------------------------------
-- [Q03] 기본 조회 : 이름에 '라면'이 들어가는 상품을 검색한다.
-- ---------------------------------------------------------------------
SELECT product_id, name, barcode, price
FROM product
WHERE name LIKE '%라면%'
ORDER BY price;

-- ---------------------------------------------------------------------
-- 실행 결과 (MySQL 8.0)
-- ---------------------------------------------------------------------
+------------+-------------------+---------------+-------+
| product_id | name              | barcode       | price |
+------------+-------------------+---------------+-------+
|          3 | 컵라면            | 8801000000035 |  1500 |
|          4 | 봉지라면 5입      | 8801000000042 |  4500 |
+------------+-------------------+---------------+-------+
