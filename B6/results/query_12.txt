-- ---------------------------------------------------------------------
-- [Q12] 서브쿼리 : 전체 상품의 평균 가격보다 비싼 상품을 확인한다.
--       괄호 안의 서브쿼리가 먼저 평균 가격을 구하고, 바깥 쿼리가 그 값과 비교한다.
-- ---------------------------------------------------------------------
SELECT product_id,
       name,
       price,
       (SELECT ROUND(AVG(price)) FROM product) AS avg_price
FROM product
WHERE price > (SELECT AVG(price) FROM product)
ORDER BY price DESC;
