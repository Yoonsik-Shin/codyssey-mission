-- ---------------------------------------------------------------------
-- [Q04] 기본 조회 : 결제 완료(PAID)된 구매 중 가장 최근 5건을 확인한다.
--       LIMIT은 MySQL 전용 문법이다.
-- ---------------------------------------------------------------------
SELECT order_id, store_id, total_amount, ordered_at
FROM orders
WHERE status = 'PAID'
ORDER BY ordered_at DESC
LIMIT 5;
